#criar index.ndx

#salvar referencia somente proteina
gmx editconf -f npt-longrun.tpr -o ref_pdb.pdb -n index.ndx <<EOF
System
EOF

Abrir o documento index.ndx
Procurar por [ Protein ] e o [ LIGANTE ] (Como quer que seu ligante esteja escrito)
Trocar o nome do ligante para Ligand, então [ Ligand ]
Pegar a parte do código da proteína e do ligante (sem os títulos)

Ctrl+S para salvar

#consertar pbc e salvar trajetoria somente proteina
gmx trjconv -f npt-longrun.trr -s npt-longrun.tpr -o md_protein.xtc -pbc nojump -n index.ndx <<EOF
System
EOF

#centralizar a proteina
gmx trjconv -f md_protein.xtc -s ref_pdb.pdb -o md_protein_fit.xtc -fit rot+trans <<EOF
Protein
System
EOF

PARA RMSF (para concatenar depois, foi o que funcionou para mim)

(100 ns pode ser mudado para qualquer momento que começou o equilíbrio)
# 1. remover saltos (nojump) a partir de 100 ns 
gmx trjconv -f npt-longrun.trr -s npt-longrun.tpr \
-o md_nojump_eq.xtc -pbc nojump -b 100000
# selecione: Protein


# 2. reconstruir molécula + centralizar
gmx trjconv -f md_nojump_eq.xtc -s npt-longrun.tpr \
-o md_center_eq.xtc -pbc mol -center
# selecione:
# Protein (center)
# Protein (output)


# 3. fitting (remover rotação/translação)
gmx trjconv -f md_center_eq.xtc -s npt-longrun.tpr \
-o md_fit_eq.xtc -fit rot+trans
# selecione:
# Protein (fit group)
# Protein (output)


# 4. RMSF
gmx rmsf -f md_fit_eq.xtc -s npt-longrun.tpr -res
# selecione: C-alpha
